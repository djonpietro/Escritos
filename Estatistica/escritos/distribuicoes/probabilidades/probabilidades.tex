\section{Cálculo das Probabilidades}

\begin{def:espaco de probabilidade}
Um espaço de medida \(\Omega, \mathcal{F}, P\) é um espaço de probabilidade se é satisfeito
que
\[
	P(\Omega) = 1.
\]
Os elementos $A \in \mathcal{F}$ são chamados de eventos.
O conjunto $\Omega$ é denominado espaço amostral.
\end{def:espaco de probabilidade}

\begin{thm:base calc prob}
Se $(\Omega, \mathcal{F}, P)$ é espaço de probabilidade, então
\begin{enumerate}
	\item[i.] $P(A^c) = 1 - P(A)$
	\item[ii.] $P(A) \leq 1$
\end{enumerate}
\begin{proof}
	Para (i), temos que $A \cup A^c = \Omega$, então pela propriedade de aditividade
	contável disjunta de medidas,
	\begin{gather*}
		P(A) + P(A^c) = P(\Omega) \\
		P(A^c) = P(\Omega) - P(A) = 1 - P(A)
	\end{gather*}
	A afirmação (iii) segue de (ii) e da propriedade de imagem não negativa
	de medida:
	\begin{gather*}
		P(A^c) \geq 0 \Longrightarrow 1 - P(A) \geq 0 \Longrightarrow P(A) \leq 1.
	\end{gather*}
\end{proof}
\end{thm:base calc prob}

\begin{thm:base2 calc prob}
\label{thm:base2 calc prob}
Se $(\Omega, \mathcal{F}, P)$ é espaço de probabilidade, e $A$ e $B$ são eventos, então
\begin{enumerate}
	\item[i] $P(B \cap A^c) = P(B) - P(B \cap A)$.
	\item[ii] $P(A \cup B) = P(A) + P(B) - P(A \cap B)$.
\end{enumerate}
\begin{proof}
	Para (i), considere primeiramente que
	\[
		(B \cap A) \cup (B \cap A^c) = B
	\]
	e como $(B \cap A) \cap (B \cap A^c) = \varnothing$, então podemos usar aditividade
	de medidas:
	\begin{gather*}
		P(B \cap A) + P(B \cap A^c) = P(B) \\
		P(B \cap A^c) = P(B) - P(B \cap A).
	\end{gather*}
	Para provar (ii), usaremos a probabilidade do complementar e a afirmação (i):
	\begin{align*}
		P(A \cup B) & = 1 - P(A^c \cap B^c)          \\
		            & = 1 - [P(A^c) - P(A^c \cap B)] \\
		            & = P(A) + P(A^c \cap B)         \\
		            & = P(A) + P(B) - P(A \cap B).
	\end{align*}
\end{proof}
\end{thm:base2 calc prob}

Para espaços amostrais enumeráveis, ao associar a cada resultado individual
um número em $[0, 1]$, de modo que a soma dos pesos resulte em 1, a soma dos pesos dos
resultados presentes num evento será a probablidade desse evento.

\begin{thm:soma prob}
\label{thm:soma prob}
Seja $\Omega = \{\omega_i\}_{\mathbb{N}}$ um conjunto enumerável, e $\mathcal{B}$ uma
$\sigma$-Álgebra de $\Omega$.
Seja $\{\delta_i\}_\nat$ uma família de funções tal que
\begin{align*}
	 & \delta_i(x) = \begin{cases}
		                 p_i, & \text{ se }  x = \omega_i \\
		                 0,   & \text{ caso contrário }
	                 \end{cases} \\
	 & \sum_{i=1}^\infty \delta_i(x) = 1
\end{align*}
Presuma também que, se $A \in \mathcal{B}$, então
\[
	P(A) = \sum_{\omega_i \in A} \delta_i(\omega_i)
\]
A tripla $(\Omega, \mathcal{B}, P)$ é um espaço de probabilidade.
\begin{proof}
	Precisamos provar que $P$ é uma medida de probabilidade que satisfaz os Axiomas de
	Medida.

	\paragraph{Não negatividade} a soma de números não negativos é não negativa, então
	para todo $A$ em $\mathcal{B}$ vale que $P(A) \geq 0$.

	\paragraph{Probablidade do espaço amostral} a série $\sum_{i=1}^\infty \delta_i(x)$
	deve coicidir com a probabilidade de $\Omega$, o que é satisfeito por hipótese.

	\paragraph{Soma de eventos disjuntos} seja $\{A_j\}_{\mathbb{N}}$ uma família
	disjunta de eventos de $\mathcal{B}$ tal que
	\[
		A_j = \{\omega_i \mid i \in I_j\}
	\]
	em que $I_j$ é um conjunto de índices. Desenvolvendo,
	\[
		P\left(\bigcup_{j=1}^\infty A_j\right) = \sum_{j=1}^\infty \sum_{i \in I_j} p_{ji}
		= \sum_{j=1}^\infty P(A_j)
	\]
\end{proof}
\end{thm:soma prob}

Esse resultado é útil para computar a probablidade de uma vasta gama de
experimentos aleatórios na prática. Um exemplo é o lançamento de um dado de 6
faces; se acreditamos que o dado é honesto, isto é, que o dado não é viciado em
nenhuma de suas faces, então podemos associar $1/6$ a cada face do dado. Se por
exemplo, deseja-se saber a probabilidade de obtermos um número menor ou igual a
3, basta somar as faces que satisfazem essa condição, obtendo uma probablidade
de $0.5$.
Experimentos os quais acreditamos não conterem vícios são chamados de equiprováveis.

As $\sigma$-Álgebras que equivalem ao conjunto das partes do espaço amostral são
chamadas de $\sigma$-Álgebra potência.
Quando o espaço amostral do experimento é enumerável e a sua $\sigma$-álgebra é a potência,
ocorre que cada subconjunto unitário do espaço amostral -- um resultado individual -- é um
evento do experimento.
Se o experimento é equiprovável, então todos os resultados são eventos de mesma probablidade.

\begin{def:evento equiprovavel}
Seja o espaço $(\Omega, \mathcal{B}, P)$ com $\Omega$ finito, $\mathcal{B} = \Parts{\Omega}$
e $P$ definido como no Teorema \ref{thm:soma prob}.
Chama-se esse espaço de equiprovável se, e somente se,
\[
	\forall \omega \in \Omega, P(\omega) = \frac{1}{|\Omega|}.
\]
\end{def:evento equiprovavel}

A probablidade de eventos em experimentos com resultados equiprováveis destaca-se
por ser intuitiva de calcular. A probablidade de evento será, simplesmente, a
fração dos resultados presentes nele. Esse resultado será enunciado como Corolário
do Teorema \ref{thm:soma prob}.

\begin{cor:equiprovavel}
\label{prop:equiprovavel}
A probabilidade de um evento $A \in \Parts{\Omega}$ num espaço $(\Omega, \Parts{\Omega}, P)$
equiprovável é igual a
\[
	\frac{|A|}{|\Omega|}
\]
\begin{proof}
	\[
		P(A) = |A| \cdot \frac{1}{|\Omega|} = \frac{|A|}{|\Omega|}
	\]
\end{proof}
\end{cor:equiprovavel}
