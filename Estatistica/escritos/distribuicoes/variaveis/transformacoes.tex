\section{Transformações}


\begin{def:transformacao}
Uma transformação de uma variável aleatória $X$ é toda função $g: \real \to \real$.
\end{def:transformacao}

Em experimentos aleatórios, comumente defini-se variáveis aleatórias a partir
de outras variáveis.
Dizemos então que a variável derivada é obtida por uma transformação da variável
original, consistindo da composição de uma função real mensurável, essa que, pelo
Teorema \ref{thm:funcoes mensuraveis}, também será mensurável, sendo, então,
uma variável aleatória.

\begin{def:variavel derivada}
Seja $X$ uma variável aleatória.
Se $g$ for uma transformação mensurável, então $g(X)$ é a variável aleatória definida
por $g \circ X$.
\end{def:variavel derivada}

Da nossa definição, a probabilidade de uma variável $g(X)$ assumir valores num conjunto $A$
deve ser o mesmo da imagem de valores de $g$ aplicada sobre realizações de $X$ pertencerem
a $A$.
De outro modo, podemos dizer que $P(g(X) \in A)$ é igual a probabilidade da pré-imagem
de $g$ em relação a $A$.

\begin{prop:prob transformacao}
\label{prop:prob transformacao}
Se $X$ é uma variável aleatória e $g$ uma transformação mensurável, então
\[
	P_X(g(X) \in A) = P(X \in g^{-1}(A))
\]
\begin{proof}
	\begin{align*}
		P_X(g(X) \in A) & = P_X(\{x \in \mathbb{R} \mid g(x) \in A\}) \\
		                & = P(X \in g^{-1}(A))
	\end{align*}
\end{proof}
\end{prop:prob transformacao}

Com essa proposição, uma transformação induz um novo espaço de probabilidade
$(\real, \borel, P_g)$ onde
\[
	P_g(A) = P_X(g^{-1}(A)) = P(X \in g^{-1}(A))
\]

\subsection{Transformação de Variáveis Discretas}

\begin{thm:prob transformacao discreta}
Seja $X$ uma variável discreta com suporte $\Omega_X$, função de massa igual a $p_X$ e
$g$ uma transformação mensurável de $X$ tal que $Y = g(X)$.
A variável $Y$ é discreta e sua função de massa é dada por
\[
	p_Y(y) = \sum_{x \in g^{-1}(y)} p_X(x)
\]
\begin{proof}
	Para $Y$ ser discreto, seu suporte $\Omega_Y$ deve ser enumerável.
	Da Proposição \ref{prop:prob transformacao} sabemos que
	\[
		P_Y(A) > 0 \leftrightarrow P(X \in g^{-1}(A)) > 0,
	\]
	e pela Proposição \ref{prop:intersecao suporte},
	\[
		g^{-1}(A) \cap \Omega_X \neq \varnothing.
	\]
	donde segue
	\[
		\forall y \in \Omega_Y \exists x \in \Omega_X, (g(x) = y \land y \in A)
	\]
	Logo, $\Omega_Y$ é a imagem de $\Omega_X$ sobre $g$, sendo, pois, enumerável.

	Para a afirmação sobre a função de massa de $Y$, a Proposição \ref{prop:prob transformacao}
	mostra que
	\[
		p_Y(y) = P_X(Y = y) = P(X \in g^{-1}(y)),
	\]
	e, pelo Lema
	\[
		p_Y(y) = \sum_{x \in g^{-1}(y)} p_X(x).
	\]
\end{proof}
\end{thm:prob transformacao discreta}

\subsection{Transformação de Variáveis Contínuas}

Para transformações de variáveis contínuas, concentramos-nos, primeiramente, nas
transformações monótonas, pois essas compõem uma classe mais fácil de se analisar.

\begin{thm:fda transformacao continua}
\label{thm:fda transformacao continua}
Seja $X$ uma variável aleatória contínua com distribuição $F_X$ e suporte $\Omega_X$.
Seja também $Y=g(X)$ com suporte $\Omega_Y$, em que $g$ é uma transformação monótona.
Se $g$ for crescente, então
\[
	F_{Y}(y) =  F_{X}(g^{-1}(y)).
\]
Caso contrário,
\[
	F_{Y}(y) = 1 - F_{X}(g^{-1}(y)).
\]
\begin{proof}
	Com efeito, da Proposição \ref{prop:prob transformacao} segue que,
	\[
		F_Y(y) = P_X(Y \leq y) = P(X \leq g^{-1} ((-\infty, y]).
	\]
	A função $g^{-1}$ tem a mesma monotonicidade de $g$.
	Se $g$ for crescente, então
	\[
		\{X \in g^{-1}((-\infty, y])\} = \{X \le g^{-1}(y)\}
	\]
	implicando em
	\[
		F_Y(y) = F_X(g^{-1}(y)).
	\]
	Se $g$ for decrescente, obtém-se
	\[
		\{X \in g^{-1}((-\infty, y])\} = \{X \ge g^{-1}(y)\}
	\]
	e
	\[
		F_Y(y) = P(X \ge g^{-1}(y))
	\]
	mas pelo Teorema \ref{thm:prob continua},
	\begin{align*}
		F_Y(y) & = P(X > g^{-1}(y))       \\
		       & = 1 - P(X \le g^{-1}(y)) \\
		       & = 1 - F_X(g^{-1}(y))
	\end{align*}
\end{proof}
\end{thm:fda transformacao continua}

Por sua vez, a densidade de transformações $Y=g(X)$ em casos em que a densidade de $X$
é contínua, pode ser obtidas via a Regra da Cadeia do Cálculo Diferencial, com a
condição de que a inversa de $g$ seja também contínua em $\Omega_Y$.
Caso a densidade de $X$ nem a inversa da transoformação sejam contínuas em $\Omega_X$ e
$\Omega_Y$ respectivamente, a regra da cadeia não poderia ser aplicada, complexificando
a determinação da densidade de $Y$.

\begin{thm:fdp transformacao continua}
\label{thm:fdp transformacao continua}
Seja $X$ uma variável aleatória com f.d.p. igual a $f_X$.
Seja $Y=g(X)$ uma variável obtida pela transformação $g$ e com suporte $\Omega_Y$.
Se $g$ é monótona, $f_X$ é contínua em $\Omega_X$ e $g^{-1}$ tem
derivada contínua em $\Omega_Y$, então a densiade $f_Y$ de $Y$ é dada por
\[
	f_{Y}(y) = \begin{cases}
		f_{X}(g^{-1}(y)) \cdot \left| \frac{d}{dy} g^{-1}(y)\right|, & \text{ se } y \in \Omega_{Y} \\
		0, \text{ caso contrário.}
	\end{cases}
\]
\begin{proof}
	Do Teorema \ref{thm:radon nikodym2},
	\[
		f_Y(y) = F_Y'(y) \qs
	\]
	Se $g$ é crescente, então
	\[
		F_Y(y) = F_X(g^{-1}(y))
	\]
	e, por consequência,
	\[
		f_Y(y) = [F_X(g^{-1}(y))]' \qs
	\]
	Por hipótese, $g^{-1}$ ter derivada contínua em $\Omega_Y$ implica que $g^{-1}$ é
	contínua em $\Omega_Y$. Ademais, toda distribuição é absolutamente contínua.
	Pela Regra da Cadeia, obtemos então
	\[
		f_Y(y) = F_X'(g^{-1}(y)) \cdot \frac{d}{dy}g^{-1}(y) \qs \text{ se } y \in \Omega_Y
	\]
	Por outro lado, se $g$ é decrescente, então por um desenvolvimento análogo chega-se
	a
	\[
		f_Y(y) = - F_X'(g^{-1}(y)) \cdot \frac{d}{dy}g^{-1}(y) \qs \text{ se } y \in \Omega_Y
	\]
	Como $\frac{d}{dy}g^{-1} \le 0$ nesse caso, então podemos sintetizar esse e o caso
	anterior em
	\[
		f_Y(y) = F_X'(g^{-1}(y)) \cdot \left|\frac{d}{dy}g^{-1}(y)\right| \qs \text{ se } y \in \Omega_Y
	\]
	Se $y \in \Omega_Y \Longrightarrow g^{-1}(y) \in \Omega_X$ e $f_X$ é contínua
	em $\Omega_X$ por hipótese, então $f_X$ é contínua em $g^{-1}(y)$.
	Além disso, pela hipótese de $g^{-1}$ ter derivada contínua em $\Omega_Y$, então
	$f_Y$ será contínua em $\Omega_Y$, e pelo Teorema \ref{thm:tfc qs},
	\[
		f_Y(y) = f_X(g^{-1}(y)) \cdot \left|\frac{d}{dy}g^{-1}(y)\right| \text{ se } y \in \Omega_Y
	\]
	Se $y \notin \Omega_Y$, então, pelo Teorema \ref{thm:suporte continua}, $f_Y(y) = 0$.
\end{proof}
\end{thm:fdp transformacao continua}

O teorema \ref{thm:fdp transformacao continua} permite-nos a generalizar a obtenção da
densidade de probabilidade de transformações não monótonas que podem ser decompostas em
partes monótonas. Um exemplo clássico é a função $g(x) = x^2$ definida em $\mathbb{R}$.
Essa função não é monótona, porém suas partes definidas separadamente em $(-\infty, 0)$ e
$[0, \infty)$ serão.
A densidade da variável definida pela transformação poderá, portanto, ser obtida pelo
resultado a seguir

\begin{thm:fdp transformacao nao monotona continua}
Seja $X$ uma variável aleatória com densidade igual a $f_X$ contínua em $\Omega_X$.
Seja $Y=g(X)$ a variável obtida pela transformação $g$ e com suporte $\Omega_Y$.
Se $\Omega_X$ admite uma partição em intervalos $\{A_i\}_{i=1}^n$ que torne possível
definir a família de funções $\{g_i: A_i \to \mathbb{R}\}_{i=1}^n$ tal que
\begin{enumerate}
	\item[i] $g_i(x) = g(x)$
	\item[ii] $g_i(x)$ é monótona em $A_i$
	\item[iii] $g_i(A_i) = \Omega_Y$ para todo $i \in [n]$.
	\item[iv] $g_i^{-1}$ tem derivada contínua em $\Omega_Y$ para todo $i \in [n]$.
\end{enumerate}
então a densidade $f_Y$ de $Y$ é dada por
\[
	f_{Y}(y) = \begin{cases}
		\sum_{i=1}^nf_{X}(g_i^{-1}(y)) \cdot \left| \frac{d}{dy} g_i^{-1}(y)\right|, & \text{ se } y \in \Omega_{Y} \\
		0, \text{ caso contrário.}
	\end{cases}
\]
\begin{proof}
	Com efeito, $f_Y = F_Y'$ e também
	\[
		F_Y(y) = P(Y \le y) = P(X \le g^{-1}((-\infty, y]))
	\]
	Mas pelo \nameref{thm:probabilidade total},
	\[
		F_Y(y) = \sum_{i=1}^n P(X \le g^{-1}((-\infty, y]) \mid X \in A_i) \cdot P(X \in A_i)
	\]
	e tomando $f(y) = F_Y'(y) \qs$,
	\[
		f_Y(y) = \sum_{i=1}^n \frac{d}{dy} F_X(g_i^{-1}(y)) + \sum_{i=1}^n \frac{d}{dy} P(X \in A_i) \qs
	\]
	como $P(X \in A_i)$ não depende de $y$, $\frac{d}{dy} P(X \in A_i) = 0$ para todo $i \in [n]$.
	Além disso, pela hipótese (iv.) e o fato de que $F_X$ é absolutamente contínua,
	se $y \in \Omega_Y$, então a equação $F_X(g_i^{-1}(y))$ será contínua em $y$.
	Da Regra da Cadeia obtemos,
	\[
		f_Y(y) = \sum_{i=1}^n f_X(g_i^{-1}(y)) \cdot \left|\frac{d}{dy}g_i^{-1}(y)\right| \qs \text{ se } y \in \Omega_Y
	\]
	Pelos Teoremas \ref{thm:tfc qs} e \ref{thm:fdp transformacao continua}, seguirá então,
	\[
		f_Y(y) = \sum_{i=1}^n f_X(g_i^{-1}(y)) \cdot \left|\frac{d}{dy}g_i^{-1}(y)\right| \text{ se } y \in \Omega_Y
	\]
	Noutro caso de $y \notin \Omega_Y$, o Teorema \ref{thm:suporte continua} implicará que
	$f_Y(y) = 0$.
\end{proof}
\end{thm:fdp transformacao nao monotona continua}
