\section{Variáveis Aleatórias Discretas}

\begin{def:variavel discreta}
Se $X$ é uma variável aleatória, então $X$ é discreta se, e somente se, seu suporte
seja enumerável.
\end{def:variavel discreta}

% Adcionar um desenvolvimento do suporte de X enquanto um conjunto discreto
%
\begin{prop:ep discreto}
Seja $X$ uma variável aleatória com suporte $\Omega_X$.
Se $\Omega_X$ é enumerável e $p_X: \mathbb{R} \to \mathbb{R}$ é tal que
\[
	p_X(x) = P(X = x),
\]
então $(\Omega_X, \Parts{\Omega_X}, p_X)$ é um espaço de probabilidade.
\begin{proof}
	Com efeito, basta provarmos que para todo $x \in \Omega_X$ temos $p_x(x) \geq 0$ e
	$\sum_{x \in \Omega_X} p_X(x) = 1$, permitindo-nos aplicar o Teorema \ref{thm:soma prob}.
	A não negatividade é satisfeita pela definição de $p_X$.
	Ademais, para todo $x$, os eventos $\{X = x\}$ são disjuntos, e pelo Axioma III de
	Kolmogorov
	\[
		\sum_{x \in \Omega_X} p_X(x) = P_X(\Omega_X)
	\]
	e pela Proposição \ref{prop:suporte}
	\[
		\sum_{x \in \Omega_X} p_X(x) = 1
	\]
\end{proof}
\end{prop:ep discreto}

Observemos então que o espaço induzido por uma variável aleatória discreta equivale
a $(\Omega_X, \Parts{\Omega_X}, p_X)$, onde a medida $p_X$ é chamada de \textbf{função
	de massa de probabilidade}, pois ela associa a cada ponto no suporte da variável uma
medida da probabilidade da variável assumir tal valor.

\begin{def:fmp}
Seja $X: \Omega \to \mathbb{R}$ uma v.a. discreta. A função $p_X: \mathbb{R} \to \mathbb{R}$
é a função de massa de probabilidade de $X$ se, e somente se,
\[
	P(X = x) = p_X(x)
\]
\end{def:fmp}
\begin{prop:fda discreto}
Se $X: \Omega \to \mathbb{R}$ é uma v.a. discreta com FMP igual a $p_X$ e suporte $\Omega_X$,
então a FDA de $X$ é
\[
	F_X(x) = \sum_{\substack{a \in \Omega_X \\ a \leq x}} p_X(a)
\]
\begin{proof}
	Suponha que $\{a_i\}_\nat$ seja uma ordenação de $\Omega_X$ tal que $x = a_k$.
	Por definição de FDA,
	\[
		F_X(x) = P(X \leq x) = P\left(\bigcup_{i=1}^k \{X = a_i\}\right)
	\]
	Como a família de eventos $\{X = x_i\}$ é disjunta, então pelo Axioma III de Kolmogrov,
	\[
		F_X(x) = \sum_{i=1}^k P(X = a_i) = \sum_{i=1}^k p_X(a_i)
	\]
	que pode ser rescrita como
	\[
		F_X(x) = \sum_{\substack{a \in \Omega_X \\ a \leq x}} p_X(a)
	\]
\end{proof}
\end{prop:fda discreto}

Com isso, se $S = \{x \in \real \mid p_X(x) > 0\} \subseteq \Omega_X$, então seja
$\{x_i\}_\nat$ uma ordenação de $S$ de tal modo
\[
	F(x) = \begin{cases}
		0,      & \text{ se } x < x_1              \\
		F(x_1), & \text{ se } x_1 \leq x < x_2     \\
		\vdots                                     \\
		F(x_i), & \text{ se } x_i \leq x < x_{i+1} \\
		\vdots                                     \\
	\end{cases}.
\]
A função $F$ é, pois, uma função constante por partes que sempre incluem a extremidade
esquerda de seu domínio devido à propriedade necessária de uma FDA sempre ser contínua
à direita. Uma vez que
\begin{gather*}
	P(X = x_i) = P(X \leq x_i) - P(X < x_i)\\
	P(X = x_i) = F(x_i) - \lim_{x \to x_i^-} F(x)\\
	P(X = x_i) = F(x_i) - F(x_{i-1}),
\end{gather*}
então $p_X(x_i)$ é o salto entre duas partes consecutivas da FDA de $X$.

Exemplos de variáveis aleatórias discretas incluem
\begin{itemize}
	\item Número de caras obtidas em 10 lançamentos.
	\item Soma dos lados obtidos no lançamento de dados.
	\item Número de clientes em uma loja em um dia.
\end{itemize}
No primeiro exemplo, as realizações possíveis são os inteiros de 0 a 10. Nos dois seguintes,
são todos os naturais. Todos esses são conjunto enumeráveis, cujos pontos isoladamente
concentram massa de probabilidade, isto é, um valor efetivo de probabilidade associado à
ao evento da variável de ter realização naquele ponto.
