\section{Variáveis Aleatórias Discretas}

\begin{def:variavel discreta}
Se $X$ é uma variável aleatória, então $X$ é discreta se, e somente se, existe um
conjunto $S \subset \real$ enumerável tal que
\[
	P(X \in S) = 1
\]
\end{def:variavel discreta}

\begin{thm:suporte discreta}
O suporte $\Omega_X$ de uma variável aleatória $X$ é enumerável se, e somente se,
$X$ é enumerável.
\begin{proof}
	Sendo $\Omega_X$ enumerável, então pelo Teorema \ref{thm:prob suporte} segue que $X$
	é discreta. Por outro lado, supondo $X$ discreta, então seja $S$ tal que $P_X(S) = 1$.
	Sendo $x \in \Omega_X$, então
	\[
		\forall \epsilon > 0, P(|X - x| < \epsilon) > 0
	\]
	mas se $x \notin S$, segue da Proposição \ref{prop:intersecao suporte} que
	\[
		\forall \epsilon > 0, S \cap (x-\epsilon, x+\epsilon) \ne \varnothing,
	\]
	concluindo-se que todo ponto em $\Omega_X \setminus S$ é de acumulação em $S$.
	Definamos
	\[
		A = S \cup \{x \in \real \setminus S \mid x \text{ é ponto de acumulação de } S\}
	\]
	donde teremos $\Omega_X \subset A$.
	Como $S$ é enumerável, então o conjunto de seus pontos de acumulação também será, assim
	como $A$.
	Uma vez que $\Omega_X$ é subconjunto de um enumerável, então ele próprio é enumerável.
\end{proof}
\end{thm:suporte discreta}

\begin{def:fmp}
Seja $X: \Omega \to \mathbb{R}$ uma v.a. discreta. A função $p_X: \mathbb{R} \to [0, 1]$
é a função de massa de probabilidade de $X$ se, e somente se,
\[
	P(X = x) = p_X(x)
\]
\end{def:fmp}

\begin{prop:fmp}
Seja $(\Omega,\mathcal F,P)$ um espaço de probabilidade e seja $X$ uma variável aleatória
discreta tal que $P(X \in S)=1$, onde $S \subset \mathbb R$. Defina
\begin{gather*}
	p_X:\mathbb R \to [0,1]\\
	x \mapsto P(X = x)
\end{gather*}
Então a aplicação
\begin{gather*}
	\mu_X:\mathcal P(S)\to [0,1]\\
	A \mapsto \sum_{x\in A} p_X(x)
\end{gather*}
é uma medida de probabilidade em $(S,\mathcal P(S))$.
\begin{proof}
	Para todo $A\subset S$,
	\[
		\mu_X(A) = \sum_{x\in A} p_X(x) = \sum_{x\in A} P(X=x).
	\]

	Como os eventos $\{X=x\}$, com $x\in A$, são dois a dois disjuntos,
	segue da $\sigma$-aditividade de $P$ que
	\[
		\mu_X(A) = P(X\in A).
	\]

	Em particular,
	\[
		\mu_X(S) = P(X\in S) = 1.
	\]

	Logo, $\mu_X$ é uma medida de probabilidade em
	$(S,\mathcal P(S))$.
\end{proof}
\end{prop:fmp}

\begin{lem:prob conjunto enumeravel}
\label{lem:prob conjunto enumeravel}
Se $X$ é variável discreta e $A \in \borel$ é enumerável, então
\[
	P(X \in A) = \sum_{x \in A} p_X(x)
\]
\begin{proof}
	Por $\sigma$-aditividade,
	\[
		P(X \in A) = \sum_{x \in A} P(X = x) = \sum_{x \in A} p_X(x)
	\]
\end{proof}
\end{lem:prob conjunto enumeravel}

\begin{prop:fda discreto}
Se $X: \Omega \to \mathbb{R}$ é uma v.a. discreta com função de massa igual a $p_X$ e
suporte $\Omega_X$, então a distribuição de $X$ é
\[
	F_X(x) = \sum_{\substack{a \in \Omega_X \\ a \leq x}} p_X(a)
\]
\begin{proof}
	Suponha $A = \{a \in \Omega_X \mid a \le x\}$ e $I = (-\infty, x]$.
	Observe que,
	\[
		P(X \le x) = P(X \in I)
	\]
	e como $A \subset I$, então por $\sigma$-aditividade,
	\[
		P(X \le x) = P(X \in A) + P(X \in I \setminus A)
	\]
	Suponha por contradição que $P(X \in I \setminus A) \ne 0$, e pela proposição
	\ref{prop:intersecao suporte}, $(I \setminus A) \cap \Omega_X \ne \varnothing$,
	o que equivale a
	\[
		I \cap (\Omega_X \setminus A) \ne \varnothing
	\]
	mas $\Omega_X \setminus A = \{a \in \Omega_X \mid a > x\}$, o que contradiz
	a hipotese anterior.
	Portanto,
	\[
		P(X \le x) = P(X \in A),
	\]
	e pelo lema \ref{lem:prob conjunto enumeravel},
	\[
		F_X(x) = \sum_{x \in A} p_X(x),
	\]
	que equivale a
	\[
		F_X(x) = \sum_{\substack{a \in \Omega_X \\ a \leq x}} p_X(a).
	\]
\end{proof}
\end{prop:fda discreto}

Com isso, se $S = \{x \in \real \mid p_X(x) > 0\} \subseteq \Omega_X$, então seja
$\{x_i\}_\nat$ uma ordenação de $S$ de tal modo
\[
	F(x) = \begin{cases}
		0,      & \text{ se } x < x_1              \\
		F(x_1), & \text{ se } x_1 \leq x < x_2     \\
		\vdots                                     \\
		F(x_i), & \text{ se } x_i \leq x < x_{i+1} \\
		\vdots                                     \\
	\end{cases}.
\]
A função $F$ é, pois, uma função constante por partes que sempre incluem a extremidade
esquerda de seu domínio devido à propriedade necessária de uma distribuição sempre ser contínua
à direita. Uma vez que
\begin{gather*}
	P(X = x_i) = P(X \leq x_i) - P(X < x_i)\\
	P(X = x_i) = F(x_i) - \lim_{x \to x_i^-} F(x)\\
	P(X = x_i) = F(x_i) - F(x_{i-1}),
\end{gather*}
então $p_X(x_i)$ é o salto entre duas partes consecutivas da distribuição de $X$.
